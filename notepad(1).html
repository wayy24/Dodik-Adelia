    <script>
        const bgMusic = document.getElementById('bg-music');
        const musicIcon = document.getElementById('music-icon');
        let isPlaying = false;

        function openInvitation() {
            document.getElementById('cover').classList.add('-translate-y-full', 'opacity-0');
            document.getElementById('main-content').classList.remove('opacity-0');
            document.getElementById('music-control').classList.remove('hidden');
           
            // Auto-play music
            bgMusic.play();
            isPlaying = true;
        }

        function toggleMusic() {
            if (isPlaying) {
                bgMusic.pause();
                musicIcon.classList.replace('fa-music', 'fa-pause');
            } else {
                bgMusic.play();
                musicIcon.classList.replace('fa-pause', 'fa-music');
            }
            isPlaying = !isPlaying;
        }

        // Countdown Timer Logic
        const targetDate = new Date("October 24, 2026 08:00:00").getTime();
        setInterval(() => {
            const now = new Date().getTime();
            const distance = targetDate - now;

            document.getElementById("days").innerText = Math.floor(distance / (1000 * 60 * 60 * 24));
            document.getElementById("hours").innerText = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            document.getElementById("minutes").innerText = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
            document.getElementById("seconds").innerText = Math.floor((distance % (1000 * 60)) / 1000);
        }, 1000);
    </script>
